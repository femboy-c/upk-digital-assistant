ЗАПУСК:
cd /home/mamai/upk
.venv/bin/uvicorn app.main:app --host 127.0.0.1 --port 8000
nohup ~/.local/bin/cloudflared tunnel --url http://127.0.0.1:8000 --no-autoupdate > /tmp/opencode/upk-tunnel.log 2>&1 &
